const express = require("express");
const http = require("http");
const { Server } = require("socket.io");

const app = express();
const server = http.createServer(app);
const io = new Server(server, {
  cors: { origin: "*" }
});

io.on("connection", (socket) => {
  console.log("user connected");

  socket.on("join-live", (room) => {
    socket.join(room);
  });

  socket.on("live-comment", (data) => {
    io.to(data.room).emit("live-comment", data);
  });

  socket.on("live-like", (data) => {
    io.to(data.room).emit("live-like", data);
  });
});

server.listen(3000, () => {
  console.log("live ads server running");
});

2) ملف السيرفر server.js
const express = require("express");
const http = require("http");
const cors = require("cors");
const { Server } = require("socket.io");

const app = express();
app.use(cors());

const server = http.createServer(app);
const io = new Server(server, {
  cors: { origin: "*" }
});

let viewers = 0;

io.on("connection", (socket) => {
  viewers++;

  io.emit("viewers", viewers);

  socket.on("join-live", (room) => {
    socket.join(room);
  });

  socket.on("comment", (data) => {
    io.to(data.room).emit("comment", data);
  });

  socket.on("like", (data) => {
    io.to(data.room).emit("like", data);
  });

  socket.on("gift", (data) => {
    io.to(data.room).emit("gift", data);
  });

  socket.on("disconnect", () => {
    viewers--;
    io.emit("viewers", viewers);
  });
});

server.listen(3000, () => {
  console.log("TikTok Live v2 running");
});
